Constructor list string and wildcard patterns
================================================================================

def respond (String -> String) \{
    match {
        Ok(value) -> value
        Err(error as e) -> e
        [] -> ""
        [head ... tail] -> head
        "" -> "empty"
        "say " ... rest -> rest
        0i64 -> "zero"
        boolean::True -> "t"
        * -> "other"
    }
}
